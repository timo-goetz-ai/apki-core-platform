### main.py
```python
from fastapi import FastAPI, WebSocket, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from contextlib import asynccontextmanager
import logging

from app.api import agents, models, prompts, tasks, automations
from app.services.agent_service import AgentManager
from app.services.task_queue import TaskQueue
from app.config.settings import Settings

logger = logging.getLogger(__name__)
settings = Settings()

# Global instances
agent_manager: AgentManager = None
task_queue: TaskQueue = None

@asynccontextmanager
async def lifespan(app: FastAPI):
    """Startup and shutdown events"""
    global agent_manager, task_queue
    
    # Startup
    logger.info("🚀 Starting AI Control Center")
    agent_manager = AgentManager()
    task_queue = TaskQueue()
    
    await agent_manager.initialize()
    await task_queue.start()
    
    yield
    
    # Shutdown
    logger.info("🛑 Shutting down AI Control Center")
    await task_queue.stop()

# Initialize FastAPI
app = FastAPI(
    title="AI Control Center",
    description="Central orchestration point for AI agents and models",
    version="1.0.0",
    lifespan=lifespan
)

# CORS
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# Include routers
app.include_router(agents.router, prefix="/api/agents", tags=["agents"])
app.include_router(models.router, prefix="/api/models", tags=["models"])
app.include_router(prompts.router, prefix="/api/prompts", tags=["prompts"])
app.include_router(tasks.router, prefix="/api/tasks", tags=["tasks"])
app.include_router(automations.router, prefix="/api/automations", tags=["automations"])

@app.get("/health")
async def health_check():
    """System health check"""
    return {
        "status": "healthy",
        "agents_running": len(agent_manager.active_agents),
        "tasks_pending": task_queue.pending_count,

@app.websocket("/ws/dashboard")
```

### agents.py (API Endpoints)
```python
from fastapi import APIRouter, HTTPException, BackgroundTasks
from pydantic import BaseModel
from typing import List, Optional
import uuid

router = APIRouter()

class AgentExecutionRequest(BaseModel):
    agent_id: str
    task_description: str
    parameters: dict = {}
    urgency: str = "normal"  # low, normal, high

class AgentStatus(BaseModel):
    agent_id: str
    name: str
    status: str
    model: str
    active_tasks: int
    total_executions: int
    success_rate: float
    last_execution: Optional[str]

@router.get("/", response_model=List[AgentStatus])
async def list_agents():
    """List all available agents"""
    agents = await agent_manager.get_all_agents()
    return [a.to_status() for a in agents]

@router.post("/execute")
async def execute_agent(
    request: AgentExecutionRequest,
    background_tasks: BackgroundTasks
):
    """Execute a specific agent"""
    
    # Create task
    task_id = str(uuid.uuid4())
    task = Task(
        id=task_id,
        agent_id=request.agent_id,
        description=request.task_description,
        parameters=request.parameters,
        priority=request.urgency,
        status="pending"
    )
    
    # Queue task
    await task_queue.add(task)
    
    # Execute in background
    background_tasks.add_task(
        agent_manager.execute_agent_task,
        request.agent_id,
        task
    )
    
    return {
        "task_id": task_id,
        "status": "queued",
        "message": f"Agent {request.agent_id} task queued"
    }

@router.get("/{agent_id}/status")
async def get_agent_status(agent_id: str):
    """Get agent status and metrics"""
    agent = await agent_manager.get_agent(agent_id)
    if not agent:
        raise HTTPException(status_code=404, detail="Agent not found")
    
    return {
        "agent_id": agent_id,
        "status": agent.status,
        "model": agent.model,
        "active_tasks": len(agent.active_tasks),
        "total_executions": agent.total_executions,
        "success_rate": agent.success_rate,
        "uptime": agent.get_uptime(),
        "metrics": {
            "avg_execution_time": agent.avg_execution_time,
            "total_tokens_used": agent.total_tokens_used,
            "errors": agent.error_count
        }
    }
```

## 💡 Phase 9: Konkrete Build-Ideen & Use Cases

### Use Case 1: Research Agent für Marktanalyse

**Workflow:**
```
1. User triggers: "Research AI market trends Q1 2026"
2. Control API assigns task to Research Agent
3. Model Router selects: deepseek-reasoner
4. Agent executes:
   - Web research via Browserless
   - Data gathering from APIs
   - Analysis & synthesis
5. Output saved: /workspace/research/market-analysis-2026-q1.md
6. Git auto-commit mit AI-generierter Message
7. Grafana Dashboard zeigt Completion
```

**Implementation:**
```python
# services/research_agent.py

class ResearchAgent:
    def __init__(self, model_router, browserless_client):
        self.model = model_router.get_model('research')
        self.browser = browserless_client
        
    async def execute(self, task):
        # 1. Determine research scope
        scope = await self._analyze_task(task)
        
        # 2. Web research
        sources = await self.browser.search_and_extract(
            query=scope['query'],
            pages=10
        )
        
        # 3. Call LLM for analysis
        analysis = await self.model.execute(
            prompt=self._build_research_prompt(sources),
            system_prompt=RESEARCH_SYSTEM_PROMPT
        )
        
        # 4. Format output
        report = self._format_report(analysis, sources)
        
        # 5. Save & commit
        await self._save_and_commit(report, task)
        
        return report
```
